📜 Script SQL Tạo Bảng Cho Supabase

Vào trang Supabase Dashboard -> Chọn dự án -> SQL Editor -> dán mã dưới đây và bấm RUN: 

-- 1. Bảng Cấu hình ứng dụng
CREATE TABLE IF NOT EXISTS app_settings (
    key TEXT PRIMARY KEY,
    shuttle_price INTEGER DEFAULT 26000,
    bank_config JSONB DEFAULT '{"bankId":"MB","accountNo":"","accountName":"","sepayToken":""}'::jsonb,
    admin_pin TEXT DEFAULT '1234',
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tự động thêm cột admin_pin nếu bảng đã tồn tại từ trước
ALTER TABLE app_settings ADD COLUMN IF NOT EXISTS admin_pin TEXT DEFAULT '1234';

-- 2. Bảng Người chơi
CREATE TABLE IF NOT EXISTS players (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT UNIQUE NOT NULL,
    penalties NUMERIC DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Bảng Lịch sử trận đấu
CREATE TABLE IF NOT EXISTS match_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    mode TEXT NOT NULL,
    winners TEXT NOT NULL,
    losers TEXT NOT NULL,
    penalty NUMERIC NOT NULL,
    money NUMERIC NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tắt RLS để cho phép đọc/ghi dữ liệu từ ứng dụng web không bị chặn
ALTER TABLE app_settings DISABLE ROW LEVEL SECURITY;
ALTER TABLE players DISABLE ROW LEVEL SECURITY;
ALTER TABLE match_history DISABLE ROW LEVEL SECURITY;

-- Bật Realtime cho cả 3 bảng
ALTER PUBLICATION supabase_realtime ADD TABLE app_settings, players, match_history;
